```python
"""GPU/CPU 能力探测层:探测一次并缓存,供 ffcompose(编码器)/engine(fp16)/jobs(并行度)复用。
探测失败一律安全降级到 CPU/libx264(无 GPU 必须仍能跑)。"""
from __future__ import annotations
import functools, os, subprocess
from dataclasses import dataclass, field
from typing import List


def _has_cuda() -> bool:
    try:
        import torch
        return bool(torch.cuda.is_available())
    except Exception:
        return False


def _gpu_info():
    try:
        import torch
        if torch.cuda.is_available():
            p = torch.cuda.get_device_properties(0)
            return p.name, int(p.total_memory // (1024 * 1024))
    except Exception:
        pass
    return "", 0


def _ffmpeg_exe() -> str:
    try:
        import imageio_ffmpeg
        return imageio_ffmpeg.get_ffmpeg_exe()
    except Exception:
        return "ffmpeg"


def _nvenc_available(exe: str) -> bool:
    try:
        out = subprocess.run([exe, "-hide_banner", "-encoders"],
                             capture_output=True, text=True, timeout=10).stdout
        return "h264_nvenc" in out
    except Exception:
        return False


@dataclass
class ComputePlan:
    has_cuda: bool
    gpu_name: str
    vram_mb: int
    cpu_count: int
    ffmpeg_exe: str
    video_encoder: str
    encoder_opts: List[str]
    infer_fp16: bool


@functools.lru_cache(maxsize=1)
def get_compute() -> ComputePlan:
    exe = _ffmpeg_exe()
    cuda = _has_cuda()
    name, vram = _gpu_info()
    nvenc = cuda and _nvenc_available(exe)
    if nvenc:
        enc, opts = "h264_nvenc", ["-c:v", "h264_nvenc", "-preset", "p4", "-rc", "vbr", "-cq", "23", "-pix_fmt", "yuv420p"]
    else:
        enc, opts = "libx264", ["-c:v", "libx264", "-preset", "veryfast", "-crf", "23", "-pix_fmt", "yuv420p"]
    return ComputePlan(has_cuda=cuda, gpu_name=name, vram_mb=vram,
                       cpu_count=os.cpu_count() or 1, ffmpeg_exe=exe,
                       video_encoder=enc, encoder_opts=opts, infer_fp16=cuda)
```

- [ ] **Step 3: 实现 build_filter**(核心:为每层一个输入,scale → 透明度 → overlay 链;层按 z 升序;时间窗加 enable)
```python
from __future__ import annotations
from typing import Any, Dict, List, Tuple

_VIDEO_EXT = {".mp4", ".mov", ".webm", ".mkv", ".avi"}


def build_filter(scene: Dict[str, Any]) -> Tuple[str, List[Dict[str, Any]], str, str]:
    W, H = int(scene["width"]), int(scene["height"])
    dur = max(0.1, int(scene["duration_ms"]) / 1000.0)
    layers = sorted(scene.get("layers") or [],
                    key=lambda l: (l.get("z") if l.get("z") is not None else 0))
    inputs: List[Dict[str, Any]] = []
    parts: List[str] = []
    # 底:黑色画布
    parts.append(f"color=c=black:s={W}x{H}:d={dur}[bg]")
    cur = "bg"
    for idx, ly in enumerate(layers):
        ipt = {"path": ly["src"], "loop": bool(ly.get("loop")) and ly.get("kind") == "video"}
        in_idx = len(inputs)
        inputs.append(ipt)
        lbl_src = f"{in_idx}:v"
        w = max(1, int(round(float(ly.get("scale") or 1.0) * W)))
        x = int(round(float(ly.get("x") or 0.0) * W))
        y = int(round(float(ly.get("y") or 0.0) * H))
        chain = f"[{lbl_src}]scale={w}:-2"
        op = ly.get("opacity")
        if op is not None and float(op) < 0.999:
            chain += f",format=yuva420p,colorchannelmixer=aa={float(op)}"
        out_lbl = f"l{idx}"
        chain += f"[{out_lbl}]"
        parts.append(chain)
        enable = ""
        wins = ly.get("windows")
        if wins:
            conds = "+".join(f"between(t,{wnd['start_ms']/1000.0},{wnd['end_ms']/1000.0})" for wnd in wins)
            enable = f":enable='{conds}'"
        nxt = f"o{idx}"
        parts.append(f"[{cur}][{out_lbl}]overlay={x}:{y}{enable}[{nxt}]")
        cur = nxt
    # 字幕(.ass 路径由 caller 生成好放 scene['subtitle']['ass_path'])
    sub = scene.get("subtitle")
    if sub and sub.get("ass_path"):
        ass = sub["ass_path"].replace("\\", "/").replace(":", "\\:")
        parts.append(f"[{cur}]subtitles='{ass}'[vout]")
        vlabel = "vout"
    else:
        vlabel = cur
    return ";".join(parts), inputs, vlabel, "aud"
```

- [ ] **Step 1-3: 实现 render**(把 inputs 转 `-i`(loop 视频加 `-stream_loop -1`)、audio 输入、filter_complex、map、encoder_opts、`-t dur`、`-progress`;失败 raise)
```python
import os, subprocess
from chayuan.server.video_generation.compute import get_compute

def render(scene: dict, out_path: str, on_progress=None) -> str:
    plan = get_compute()
    fc, inputs, vlabel, _ = build_filter(scene)
    cmd = [plan.ffmpeg_exe, "-y", "-loglevel", "error"]
    for ipt in inputs:
        if ipt.get("loop"):
            cmd += ["-stream_loop", "-1"]
        cmd += ["-i", ipt["path"]]
    cmd += ["-i", scene["audio_path"]]          # 最后一个输入=音频
    aidx = len(inputs)
    cmd += ["-filter_complex", fc,
            "-map", f"[{vlabel}]", "-map", f"{aidx}:a",
            "-t", str(max(0.1, scene["duration_ms"] / 1000.0)),
            "-r", "25", "-c:a", "aac", *plan.encoder_opts, out_path]
    subprocess.run(cmd, check=True)
    return out_path
```
